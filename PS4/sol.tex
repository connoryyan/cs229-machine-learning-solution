\begin{homeworkProblem}
\begin{enumerate}
    \item[(a)] 
    \[
    \begin{aligned}
        \E_{\substack{s\sim p(s) \\ a\sim\pi_0(s, a)}}\left[
            \frac{\pi_1(s, a)}{\pi_0(s, a)}
            R(s, a)
        \right]
        &= \sum_{(s, a)} {
            \frac{\pi_1(s, a)}{\pi_0(s, a)}
            R(s, a)
            p(s)
            \pi_0(s, a)
        } \\
        &= \sum_{(s, a)} {
            \pi_1(s, a)
            R(s, a)
            p(s)
        } \\
        &= \E_{\substack{s\sim p(s) \\ a\sim\pi_1(s, a)}}\left[
            R(s, a)
        \right]
    \end{aligned}
    \]

    \item[(b)]
    \[
    \begin{aligned}
        \frac{
            \E_{\substack{s\sim p(s) \\ a\sim\pi_0(s, a)}}\left[
                \frac{\pi_1(s, a)}{\pi_0(s, a)}
                R(s, a)
            \right]
        }{
            \E_{\substack{s\sim p(s) \\ a\sim\pi_0(s, a)}}\left[
                \frac{\pi_1(s, a)}{\pi_0(s, a)}
            \right] 
        }
        &= \frac{
            \sum_{(s, a)}{
                \frac{\pi_1(s, a)}{\pi_0(s, a)}
                R(s, a)
                p(s)
                \pi_0(s, a)
            }
        }{
            \sum_{(s, a)}{
                \frac{\pi_1(s, a)}{\pi_0(s, a)}
                p(s)
                \pi_0(s, a)
            }
        } \\
        &= \frac{
            \sum_{(s, a)}{
                R(s, a)
                p(s)
                \pi_1(s, a)
            }
        }{
            \sum_{(s, a)}{
                p(s)
                \pi_1(s, a)
            }
        } \\
        &= \E_{\substack{s\sim p(s) \\ a\sim\pi_1(s, a)}}\left[
            R(s, a)
        \right]
    \end{aligned}
    \]

    \item[(c)] Suppose we have $n$ samples. Then the weighted importance sampling estimator has the form
    \[
        \frac{
            \sum_{i=1}^n {
                \frac{\pi_1(s_i, a_i)}{\pi_0(s_i, a_i)}
                R(s_i, a_i)
            }
        }{
            \sum_{i=1}^n {
                \frac{\pi_1(s_i, a_i)}{\pi_0(s_i, a_i)}
            }
        }
    \]
    Consider the case that there is only one sample, i.e, $n = 1$.
    \[
    \begin{aligned}
        \frac{
            \sum_{i=1}^n {
                \frac{\pi_1(s_i, a_i)}{\pi_0(s_i, a_i)}
                R(s_i, a_i)
            }
        }{
            \sum_{i=1}^n {
                \frac{\pi_1(s_i, a_i)}{\pi_0(s_i, a_i)}
            }
        }
        &= \frac{
            \frac{\pi_1(s_1, a_1)}{\pi_0(s_1 a_1)}
            R(s_1, a_1)
        }{
            \frac{\pi_1(s_1, a_1)}{\pi_0(s_1, a_1)}
        } \\
        &= R(s_1, a_1)
    \end{aligned}
    \]
    The expectation is
    \[
    \begin{aligned}
        \E_{\substack{s_1\sim p(s_1) \\ a_1\sim\pi_0(s_1, a_1)}}\left[
            R(s_1, a_1)
        \right]
        &= \E_{\substack{s\sim p(s) \\ a\sim\pi_0(s, a)}}\left[
            R(s, a)
        \right]
    \end{aligned}
    \]
    Which is not quite likely to be equal to the objective value $\E_{\substack{s\sim p(s) \\ a\sim\pi_1(s, a)}}\left[ R(s, a) \right]$.

\begin{homeworkProblem}
Note that
\[
f_u(x) = u \cdot u^Tx
\]
Hence,
\[
\begin{aligned}
    \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        \| x^{(i)} - f_u(x^{(i)}) \|_2^2
    }
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        \| x^{(i)} - u \cdot u^Tx^{(i)} \|_2^2
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        \| (I - u u^T) x^{(i)} \|_2^2
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        \left( (I - u u^T) x^{(i)} \right)^T 
        (I - u u^T) x^{(i)}
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        (x^{(i)})^T (I - u u^T)^T
        (I - u u^T) x^{(i)}
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        (x^{(i)})^T 
        (I - 2uu^T + uu^Tuu^T)
        x^{(i)}
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        (x^{(i)})^T 
        (u^Tu - 2) uu^T
        x^{(i)}
    } \\
    &= \arg \min_{u:u^Tu=1} \sum_{i=1}^m {
        (x^{(i)})^T 
        (1 - 2) uu^T
        x^{(i)}
    } \\
    &= \arg \max_{u:u^Tu=1} \sum_{i=1}^m {
        (x^{(i)})^T 
        uu^T
        x^{(i)}
    } \\
    &= \arg \max_{u:u^Tu=1} \sum_{i=1}^m {
        ((x^{(i)})^T u)^2
    }
\end{aligned}
\]
Let $\mathcal{L} = \left[ \sum_{i=1}^m { ((x^{(i)})^T u)^2 } \right] - \lambda(u^Tu-1)$. Then
\[
\begin{aligned}
    \frac{\partial \mathcal{L}}{\partial u}
    &= \left[ \sum_{i=1}^m {
            2x^{(i)}(x^{(i)})^Tu
        } \right]
        - 2\lambda u
\end{aligned}
\]
Let $\frac{\partial \mathcal{L}}{\partial u} = 0$. Then
\[
\begin{aligned}
    \left[ \sum_{i=1}^m {
        2x^{(i)}(x^{(i)})^Tu
    } \right]
    - 2\lambda u &= 0 \\
    \sum_{i=1}^m {
        x^{(i)}(x^{(i)})^Tu
    } &= \lambda u
\end{aligned}
\]
Note that
\[
\begin{aligned}
    \Sigma &= \E[(x-\mu)(x-\mu)^T] \\
    &= \frac{1}{n} \sum_{i=1}^n {
        x^{(i)}(x^{(i)})^T
    }, \ (\mu = 0)
\end{aligned}
\]
Therefore, as
$\sum_{i=1}^m {
    x^{(i)}(x^{(i)})^Tu
} = \lambda u
$
\[
\begin{aligned}
    n\Sigma u &= \lambda u \\
    \Sigma u &= \frac{\lambda}{n} u
\end{aligned}
\]
That is to say, $u$ is the first eigenvector of $\Sigma$, and therefore is the first principal component.

\end{homeworkProblem}
